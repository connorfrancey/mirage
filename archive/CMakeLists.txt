cmake_minimum_required(VERSION 3.24)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(MiragePayloadArchive LANGUAGES C CXX)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  option(MIRAGE_BUILD_TESTS "Build MPAF tests" ON)
  option(MIRAGE_BUILD_BENCHMARKS "Build MPAF benchmarks" OFF)
  include(../cmake/CPM.cmake)
  include(../cmake/Versions.cmake)
  if(MIRAGE_BUILD_TESTS)
    enable_testing()
  endif()
endif()

if(NOT TARGET xxhash::xxhash)
  CPMAddPackage(NAME xxhash GITHUB_REPOSITORY Cyan4973/xxHash GIT_TAG v0.8.3 DOWNLOAD_ONLY YES)
  add_library(mirage_mpaf_xxhash INTERFACE)
  target_include_directories(mirage_mpaf_xxhash SYSTEM INTERFACE "${xxhash_SOURCE_DIR}")
  target_compile_definitions(mirage_mpaf_xxhash INTERFACE XXH_INLINE_ALL)
  add_library(xxhash::xxhash ALIAS mirage_mpaf_xxhash)
endif()

if(NOT TARGET simdutf::simdutf)
  CPMAddPackage(NAME simdutf GITHUB_REPOSITORY simdutf/simdutf GIT_TAG ${SIMDUTF_VERSION}
    OPTIONS "SIMDUTF_TESTS OFF" "SIMDUTF_BENCHMARKS OFF" "SIMDUTF_TOOLS OFF" "SIMDUTF_ICONV OFF" "BUILD_TESTING OFF")
endif()

add_library(mirage_mpaf STATIC src/format.cpp src/hash.cpp src/reader.cpp src/writer.cpp)
add_library(mirage::mpaf ALIAS mirage_mpaf)
target_compile_features(mirage_mpaf PUBLIC cxx_std_20)
target_include_directories(mirage_mpaf PUBLIC include)
target_link_libraries(mirage_mpaf PRIVATE xxhash::xxhash simdutf::simdutf)
CPMAddPackage(NAME lz4 GITHUB_REPOSITORY lz4/lz4 GIT_TAG v1.10.0 DOWNLOAD_ONLY YES)
add_library(mirage_mums_lz4 STATIC
  ${lz4_SOURCE_DIR}/lib/lz4.c
  ${lz4_SOURCE_DIR}/lib/lz4frame.c
  ${lz4_SOURCE_DIR}/lib/lz4hc.c
  ${lz4_SOURCE_DIR}/lib/xxhash.c)
target_include_directories(mirage_mums_lz4 SYSTEM PUBLIC "${lz4_SOURCE_DIR}/lib")
target_compile_definitions(mirage_mums_lz4 PRIVATE XXH_NAMESPACE=LZ4_)

add_library(mirage_mums STATIC src/mums.cpp)
add_library(mirage::mums ALIAS mirage_mums)
target_compile_features(mirage_mums PUBLIC cxx_std_20)
target_include_directories(mirage_mums PUBLIC include)
target_link_libraries(mirage_mums PRIVATE mirage::mpaf mirage_mums_lz4)
add_executable(mirage_pack src/pack.cpp)
target_link_libraries(mirage_pack PRIVATE mirage::mpaf mirage::mums)
if(WIN32)
  target_compile_definitions(mirage_mpaf PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(mirage_mpaf PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(mirage_mpaf PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(MIRAGE_BUILD_TESTS)
  if(NOT TARGET GTest::gtest_main)
    CPMAddPackage(NAME googletest GITHUB_REPOSITORY google/googletest GIT_TAG ${GOOGLETEST_VERSION}
      OPTIONS "INSTALL_GTEST OFF" "BUILD_GMOCK OFF" "gtest_force_shared_crt OFF")
  endif()
  add_executable(mirage_mpaf_tests tests/test_mpaf.cpp tests/test_mums.cpp)
  target_link_libraries(mirage_mpaf_tests PRIVATE mirage::mpaf mirage::mums GTest::gtest_main)
  target_compile_definitions(mirage_mpaf_tests PRIVATE MPAF_VECTOR_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors")
  include(GoogleTest)
  gtest_discover_tests(mirage_mpaf_tests)
  if(TARGET mirage_tests)
    add_dependencies(mirage_tests mirage_mpaf_tests)
  endif()
endif()

if(MIRAGE_BUILD_BENCHMARKS AND CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  CPMAddPackage(NAME benchmark GITHUB_REPOSITORY google/benchmark GIT_TAG ${GOOGLEBENCH_VERSION}
    OPTIONS "BENCHMARK_ENABLE_TESTING OFF" "BENCHMARK_ENABLE_GTEST_TESTS OFF"
      "BENCHMARK_ENABLE_INSTALL OFF" "BENCHMARK_ENABLE_WERROR OFF" "BENCHMARK_ENABLE_LIBPFM OFF")
  add_executable(mirage_mpaf_bench tests/bench_mpaf.cpp)
  target_link_libraries(mirage_mpaf_bench PRIVATE mirage::mpaf benchmark::benchmark_main)
endif()
